import socket
import math
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
import os

print("==================================================")
print(" INITIALIZING ML MODEL (Random Forest) ")
input_file = 'Experimental_Data_VisionApp.txt'
if not os.path.exists(input_file):
    print(f"[ERROR] Could not find '{input_file}'. Ensure dataset is present.")
    exit()

df = pd.read_csv(input_file, sep='\t')
df = df.dropna(subset=['ue_x', 'ue_y', 'SNR (dB)'])
features = ['gnb_x', 'gnb_y', 'ue_x', 'ue_y', 'obs_x', 'obs_y', 'Blockage']
X = df[features]
y = df['SNR (dB)']

ml_model = RandomForestRegressor(n_estimators=100, max_depth=10, random_state=42)
ml_model.fit(X.values, y.values)
MAX_SNR = 39.0
print(" ML MODEL READY! Dynamic Inference Active.")
print("==================================================")

UDP_IP = "127.0.0.1"
UDP_PORT = 9999
sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
sock.bind((UDP_IP, UDP_PORT))

print(" 6G VISION-APP xAPP (STRESS TEST MODE - DUAL EXPERIMENT A/B) ")
print(" Listening on Port 9999... Waiting for ns-3. ")
print("==================================================")

T_CTRL = 0.200       
MAX_SPEED = 1.5      
SAFE_MARGIN = 2.5    
OBSTACLE_RADIUS = 0.8 

current_gnb_x = 4.0

while True:
    data, addr = sock.recvfrom(1024)
    msg = data.decode('utf-8')
    parts = msg.split(',')
    
    gnb_x, gnb_y = float(parts[0]), float(parts[1])
    ue_x, ue_y = float(parts[2]), float(parts[3])
    obs_x, obs_y = float(parts[4]), float(parts[5])
    enable_xapp = int(parts[6])  

    if enable_xapp == 1:
        target_x = ue_x
        if abs(target_x - obs_x) < SAFE_MARGIN:
            if ue_x >= obs_x:
                target_x = obs_x + SAFE_MARGIN
            else:
                target_x = obs_x - SAFE_MARGIN

        max_step = MAX_SPEED * T_CTRL
        diff = target_x - current_gnb_x
        if abs(diff) <= max_step:
            current_gnb_x = target_x
        else:
            current_gnb_x += math.copysign(max_step, diff)
            
        active_gnb_x = current_gnb_x
    else:
        active_gnb_x = gnb_x
        target_x = gnb_x

    l2 = (ue_x - active_gnb_x)**2 + (ue_y - gnb_y)**2
    t = 0.0
    if l2 != 0:
        t = ((obs_x - active_gnb_x) * (ue_x - active_gnb_x) + (obs_y - gnb_y) * (ue_y - gnb_y)) / l2
        t = max(0.0, min(1.0, t))
    
    proj_x = active_gnb_x + t * (ue_x - active_gnb_x)
    proj_y = gnb_y + t * (ue_y - gnb_y)
    dist_to_los = math.sqrt((obs_x - proj_x)**2 + (obs_y - proj_y)**2)
    
    is_blocked_now = 1 if dist_to_los <= OBSTACLE_RADIUS else 0
    
    new_state = [[active_gnb_x, gnb_y, ue_x, ue_y, obs_x, obs_y, is_blocked_now]]
    predicted_snr = ml_model.predict(new_state)[0]
    
    penalty_db = MAX_SNR - predicted_snr
    if penalty_db < 0 or not is_blocked_now:
        penalty_db = 0.0  

    mode_str = "Exp B (xApp)" if enable_xapp == 1 else "Exp A (Base)"
    print(f"[{mode_str}] Target: {target_x:.2f} | Active gNB: {active_gnb_x:.2f} | Penalty: {penalty_db:.2f} dB")

    response_msg = f"{active_gnb_x},{penalty_db}"
    sock.sendto(response_msg.encode('utf-8'), addr)
