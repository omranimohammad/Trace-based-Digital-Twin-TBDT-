double GetLatestDataSinr() {
    std::ifstream file("DlDataSinr.txt");
    if (!file.is_open()) return 0.0;
    
    std::string line, lastLine;
    while (std::getline(file, line)) {
        if (!line.empty()) lastLine = line;
    }
    if (lastLine.empty()) return 0.0;
    
    std::stringstream ss(lastLine);
    std::string token;
    int col = 0;
    double sinr = 0.0;
    
    while (std::getline(ss, token, '\t')) {
        if (col == 4) { 
            try { sinr = std::stod(token); } catch(...) {}
            break;
        }
        col++;
    }
    return sinr;
}

void MonitorThroughput(Ptr<PacketSink> sink, double interval) {
    double now = Simulator::Now().GetSeconds();
    uint64_t totalRx = sink->GetTotalRx();
    uint64_t bytesReceivedThisInterval = totalRx - g_lastTotalRx;
    
    double mbps = (bytesReceivedThisInterval * 8.0) / (interval * 1000 * 1000);
    double currentSinr = GetLatestDataSinr();

    std::cout << "SIM_THROUGHPUT," << now << "," << mbps << std::endl;
    std::cout << "SIM_SNR," << now << "," << currentSinr << std::endl;
    
    if (g_fidelityCsv.is_open()) {
        g_fidelityCsv << now << "," << mbps << "," << currentSinr << "," 
                      << (g_isBlockedGlobal ? 1 : 0) << "\n";
    }

    g_lastTotalRx = totalRx;
    Simulator::Schedule(Seconds(interval), &MonitorThroughput, sink, interval);
}

void VisionAppControlLoop(Ptr<MobilityModel> gnbMob, Ptr<MobilityModel> ueMob, Ptr<MobilityModel> obsMob, bool enableXapp) {
    Vector gnbPos = gnbMob->GetPosition();
    Vector uePos = ueMob->GetPosition();
    Vector obsPos = obsMob->GetPosition();

    std::ostringstream msg;
    msg << gnbPos.x << "," << gnbPos.y << "," 
        << uePos.x << "," << uePos.y << "," 
        << obsPos.x << "," << obsPos.y << ","
        << (enableXapp ? "1" : "0");
    std::string message = msg.str();

    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd >= 0) {
        struct timeval tv;
        tv.tv_sec = 1;
        tv.tv_usec = 0;
        setsockopt(sockfd, SOL_SOCKET, SO_RCVTIMEO, (const char*)&tv, sizeof tv);

        struct sockaddr_in servaddr;
        memset(&servaddr, 0, sizeof(servaddr));
        
        servaddr.sin_family = AF_INET;
        servaddr.sin_port = htons(9999);
        inet_pton(AF_INET, "127.0.0.1", &servaddr.sin_addr);

        ssize_t sent = sendto(sockfd, message.c_str(), message.length(), 0, (const struct sockaddr *) &servaddr, sizeof(servaddr));
        if (sent > 0) {
            char buffer[1024];
            socklen_t len = sizeof(servaddr);
            int n = recvfrom(sockfd, (char *)buffer, 1024, MSG_WAITALL, (struct sockaddr *) &servaddr, &len);
            
            if (n > 0) {
                buffer[n] = '\0';
                std::string recvStr(buffer);
                size_t commaPos = recvStr.find(',');
                
                if (commaPos != std::string::npos) {
                    double x_target = std::stod(recvStr.substr(0, commaPos));
                    g_current_penalty = std::stod(recvStr.substr(commaPos + 1));
                    
                    // Strictly enforce mobility separation
                    if (enableXapp) {
                        gnbMob->SetPosition(Vector(x_target, gnbPos.y, gnbPos.z));
                    }
                }
            }
        }
        close(sockfd);
    }
    Simulator::Schedule(MilliSeconds(200), &VisionAppControlLoop, gnbMob, ueMob, obsMob, enableXapp);
}

class MlPropagationLossModel : public PropagationLossModel {
public:
    static TypeId GetTypeId(void) {
        static TypeId tid = TypeId("ns3::MlPropagationLossModel")
            .SetParent<PropagationLossModel>()
            .SetGroupName("Propagation")
            .AddConstructor<MlPropagationLossModel>();
        return tid;
    }

    MlPropagationLossModel() {}

    void SetObstacleMobility(Ptr<MobilityModel> obsMobility) {
        m_obsMobility = obsMobility;
    }

private:
    Ptr<MobilityModel> m_obsMobility;

    virtual double DoCalcRxPower(double txPowerDbm, Ptr<MobilityModel> a, Ptr<MobilityModel> b) const override {
        if (!m_obsMobility) { return txPowerDbm; }

        double now = Simulator::Now().GetSeconds();
        Vector gnbPos = a->GetPosition();
        Vector uePos = b->GetPosition();
        Vector obsPos = m_obsMobility->GetPosition();

        double l2 = std::pow(uePos.x - gnbPos.x, 2) + std::pow(uePos.y - gnbPos.y, 2);
        double t = 0.0;
        if (l2 != 0.0) {
            t = ((obsPos.x - gnbPos.x) * (uePos.x - gnbPos.x) + (obsPos.y - gnbPos.y) * (uePos.y - gnbPos.y)) / l2;
            t = std::max(0.0, std::min(1.0, t)); 
        }
        double projX = gnbPos.x + t * (uePos.x - gnbPos.x);
        double projY = gnbPos.y + t * (uePos.y - gnbPos.y);
        double distanceToLoS = std::sqrt(std::pow(obsPos.x - projX, 2) + std::pow(obsPos.y - projY, 2));

        bool isBlocked = (distanceToLoS <= OBSTACLE_RADIUS_METERS);
        g_isBlockedGlobal = isBlocked;

        double appliedPenalty = isBlocked ? g_current_penalty : 0.0;
        double finalPower = txPowerDbm - appliedPenalty;

        static int lastPrintedSecond = -1;
        int currentSecond = static_cast<int>(now);
        
        if (currentSecond != lastPrintedSecond) {
            std::cout << "==================================================" << std::endl;
            std::cout << "[Time: " << currentSecond << "s] SPATIAL TOPOLOGY:" << std::endl;
            std::cout << "  -> Tower (gNB) Loc: X=" << gnbPos.x << ", Y=" << gnbPos.y << std::endl;
            std::cout << "  -> Phone (UE)  Loc: X=" << uePos.x << ", Y=" << uePos.y << std::endl;
            std::cout << "  -> Obstacle    Loc: X=" << obsPos.x << ", Y=" << obsPos.y << std::endl;
            std::cout << "[Time: " << currentSecond << "s] RF METRICS:" << std::endl;
            std::cout << "  -> Line-of-Sight  : " << (isBlocked ? "BLOCKED (NLOS) [X]" : "CLEAR (LoS) [✓]") << std::endl;
            std::cout << "  -> ML Signal Loss : -" << appliedPenalty << " dB" << std::endl;
            std::cout << "  -> Channel Power  : " << finalPower << " dBm" << std::endl;
            lastPrintedSecond = currentSecond;
        }

        return finalPower;
    }
    virtual int64_t DoAssignStreams(int64_t stream) override { return 0; }
};

void LoadTraceData(std::string filename) {
    std::ifstream file(filename);
    std::string line;
    std::getline(file, line); 

    while (std::getline(file, line)) {
        if (line.empty()) continue;
        std::stringstream ss(line);
        std::string token;
        TraceRecord rec;
        
        int col = 0;
        while (std::getline(ss, token, ',')) {
            try {
                if (col == 0) rec.time = std::stod(token);
                if (col == 1) rec.gnb_x = std::stod(token);
                if (col == 2) rec.gnb_y = std::stod(token);
                if (col == 3) rec.ue_x = std::stod(token);
                if (col == 4) rec.ue_y = std::stod(token);
                if (col == 5) rec.obs_x = std::stod(token);
                if (col == 6) rec.obs_y = std::stod(token);
            } catch (...) {} 
            col++;
        }
        g_traceData.push_back(rec);
    }
}
